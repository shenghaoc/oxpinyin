# Steps run in order.  A scenario= line starts a new one.
#   step=type            text=<ascii>     one process_key per character
#   step=key             name=<logical>   enter | escape | space | backspace
#   step=select          index=<n>        choose by index
#   step=select-text     text=<phrase>    choose the candidate with this text
#   step=expect-preedit  text=<text>      the whole preedit text
#   step=expect-candidate index=<n>  text=<phrase>
#   step=expect-absent   text=<phrase>
#   step=expect-empty                     nothing is composing
#   step=commit          text=<text>      commit and assert what came out
